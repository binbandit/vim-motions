## Within a line

<MotionList>
  <Motion
    keys="0"
    name="First column"
    before={'    hello|'}
    after={'|    hello'}
    note="Column 1 — even if the line is indented."
  />
  <Motion
    keys="^"
    name="First non-blank character"
    before={'    hello|'}
    after={'    |hello'}
  />
  <Motion
    keys="$"
    name="End of line"
    before={'|hello world'}
    after={'hello worl|d'}
    note="Inclusive. 2$ goes to the end of the next line."
  />
  <Motion
    keys="g_"
    name="Last non-blank character"
    before={'|hello world   '}
    after={'hello worl|d   '}
    note="Ignores trailing spaces."
  />
  <Motion
    keys="|"
    name="Screen column [count]"
    before={'abcdef|'}
    after={'ab|cdef'}
    note="3| jumps to column 3 (a=1)."
  />
  <Motion keys="g0" name="First character of soft-wrapped screen line" />
  <Motion keys="g^" name="First non-blank of soft-wrapped screen line" />
  <Motion keys="g$" name="Last character of soft-wrapped screen line" />
  <Motion keys="gm" name="Middle of screen line" sameAs="g0" example="Then half of 'columns' to the right" />
  <Motion keys="gM" name="Middle of text line" />
</MotionList>

## Across the file

<MotionList>
  <Motion
    keys="gg"
    name="Line [count] (default 1)"
    also={['Ctrl-Home']}
    before={'one\ntwo\n|three'}
    after={'|one\ntwo\nthree'}
  />
  <Motion
    keys="G"
    name="Line [count] (default last)"
    before={'|one\ntwo\nthree'}
    after={'one\ntwo\n|three'}
  />
  <Motion keys="Ctrl-End" name="Go to last character of last line" sameAs="G" sameAsThen="$" />
  <Motion
    keys="{count}%"
    name="To count% through the file"
    before={'|aaaa\nbbbb\ncccc\ndddd'}
    after={'aaaa\nbbbb\n|cccc\ndddd'}
    note="50% on a 4-line file lands near the middle. Bare % without a count is matching — see Matching."
  />
  <Motion keys=":{n}" name="Ex goto line n" example=":80 then Enter" />
  <Motion keys="go" name="Goto byte [count] in the buffer" example="Also :go[to] · default 1" />
</MotionList>

## Line operators (shorthand)

<Buffer
  keys="D"
  before={'keep |delete me'}
  after={'keep| '}
  note="D is like d$ — deletes to end of line."
/>

<Buffer
  keys="C"
  before={'keep |change me'}
  after={'keep |'}
  note="C is like c$ — deletes to EOL and enters Insert."
/>

| Keys | Meaning |
| --- | --- |
| <Keys keys="dd" link /> | Delete line |
| <Keys keys="cc" link /> / <Keys keys="S" link /> | Change line |
| <Keys keys="yy" link /> | Yank line |
| <Keys keys="D" link /> | Delete to end of line (<SameAs keys="d$" />) |
| <Keys keys="C" link /> | Change to end of line (<SameAs keys="c$" />) |
| <Keys keys="I" link /> | Insert at first non-blank |
| <Keys keys="A" link /> | Append at end of line |
